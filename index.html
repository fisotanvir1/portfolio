#!/usr/bin/env python3 """ Usage: python optimize_index.py index.html Creates index.optimized.html next to the original (original is not touched). Each step reports OK or SKIPPED so you can see exactly what changed. """ import re, sys src = sys.argv[1] if len(sys.argv) > 1 else "index.html" s = open(src, encoding="utf-8").read() report = [] def step(name, ok): report.append(("OK " if ok else "SKIPPED") + " - " + name) def cut(text, start_marker, end_marker, replacement=""): """Remove text from the '/* ====' banner before start_marker up to the banner before end_marker.""" a = text.find(start_marker) b = text.find(end_marker) if a < 0 or b < 0 or b < a: return text, False a = text.rfind("/* ====", 0, a) b = text.rfind("/* ====", 0, b) return text[:a] + replacement + text[b:], True # ---------- 1. English-only: HTML ---------- n0 = len(s) s = re.sub(r'\s*', "", s, count=1, flags=re.S) step("Language dropdown removed from nav", len(s) < n0) s, c = re.subn(r'\s+data-i18n(?:-html|-placeholder)?="[^"]*"', "", s) step(f"Removed {c} data-i18n attributes", c > 0) s, ok = re.subn(r'\s*\s*
]*>
', "", s) step("Language toast element removed", ok > 0) s, ok = re.subn(r"\s*var l = localStorage\.getItem\('tanvir-lang'\);\s*if \(l\) document\.documentElement\.setAttribute\('lang', l\);", "", s) step("Language restore removed from script", ok > 0) # ---------- 2. English-only: CSS ---------- s, c = re.subn(r'\.lang-(?:wrap|select)[^{]*\{[^}]*\}\s*', "", s) step(f"Removed {c} language CSS rules", c > 0) s, ok = cut(s, "CSS SECTION: TOAST", "CSS SECTION: SCROLL REVEAL ANIMATIONS") step("Toast CSS removed", ok) # ---------- 3. English-only: JS ---------- english_block = '''/* ============================================================ JS SECTION: ENGLISH UI STRINGS (only the strings JS still needs) ============================================================ */ const STRINGS = { 'lightbox.placeholder': 'Sample placeholder \\u2014 no image file provided yet. Add a real image inside this card to display it here.', 'form.sending': 'Sending your message\\u2026', 'form.success': 'Thank you! Your message was sent successfully. I will get back to you as soon as possible.', 'form.error': 'Sorry, your message could not be sent. Please try again, or email me directly at tanvirofficial105@gmail.com.', 'form.errorNetwork': 'Network error \\u2014 your message was not sent. Please check your internet connection, or email me directly at tanvirofficial105@gmail.com.' }; function t(key) { return STRINGS[key] !== undefined ? STRINGS[key] : key; } try { localStorage.removeItem('tanvir-lang'); } catch (e) {} ''' s, ok = cut(s, "JS SECTION: TRANSLATIONS (i18n)", "JS SECTION: APPLY SITE CONFIGURATION", english_block) step("Translation dictionaries replaced by small English strings object", ok) s, ok = cut(s, "JS SECTION: LANGUAGE SWITCHER", "JS SECTION: MOBILE NAVIGATION") step("Language switcher JS removed", ok) s, c = re.subn(r"\s*document\.addEventListener\('tanvir:lang-change', function \(\) \{.*?\}\);", "", s, flags=re.S) step("Lightbox language-change listener removed", c > 0) # ---------- 4. Non-blocking CSS (Google Fonts + Font Awesome) ---------- def defer(m): tag = m.group(0) href = re.search(r'href="([^"]+)"', tag).group(1) rp = ' referrerpolicy="no-referrer"' if "referrerpolicy" in tag else "" return ('\n' '') % (href, rp, href, rp) s, c1 = re.subn(r'', defer, s) step("Google Fonts CSS no longer render-blocking", c1 > 0) s, c2 = re.subn(r']*>', defer, s) step("Font Awesome CSS no longer render-blocking", c2 > 0) # ---------- Sanity checks ---------- for leftover in ("I18N", "langSelect", "UNTRANSLATED_LANGS", "applyTranslations", "data-i18n"): if leftover in s: report.append("WARNING - '%s' still found in output; check manually" % leftover) out = src.rsplit(".", 1)[0] + ".optimized.html" open(out, "w", encoding="utf-8").write(s) print("\n".join(report)) print("\nWritten:", out)
